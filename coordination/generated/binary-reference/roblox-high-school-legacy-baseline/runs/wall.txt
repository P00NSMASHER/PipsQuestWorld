function waitfor(parent,name)
	while parent:FindFirstChild(name)==nil do
		wait()
	end
	return parent:FindFirstChild(name)
end

function RemoveFlyStuff()
	wind:Stop()
	local plr=game.Players.LocalPlayer
	if plr then
		local chr=plr.Character
		if chr then
			for i,v in ipairs(chr:getChildren()) do
				if v.Name=="Humanoid" then
					v.PlatformStand=false
				end
				if v and v.Name=="EffectCloud" then
					sp.Call:FireServer(v, 'Destroy')
				end
			end
			local torso=chr:FindFirstChild("HumanoidRootPart")
			if torso~=nil then
				for i,v in ipairs(torso:GetChildren()) do
					if v and (v.Name=="FlightGyro" or v.Name=="FlightVelocity") then
						v:remove()
					end
				end
			end
		end
	end
	handle.Transparency=0
	sp.Set:FireServer(handle.Smoke, 'Enabled', true)
	if lastanim~=nil then
		lastanim:Stop()
	end
end

function fly()		--(de)activate fly mode
	if not equipped then
		flying=false
	else
		flying=not flying
	end
	local plr=game.Players.LocalPlayer
	RemoveFlyStuff()
	if flying then
		local torso=sp.Parent:FindFirstChild("HumanoidRootPart")
		local humanoid=sp.Parent:FindFirstChild("Humanoid")
		if torso and humanoid and humanoid.Health>0 then
			sp.Set:FireServer(handle, 'Transparency', 1)
			sp.Set:FireServer(handle.Smoke, 'Enabled', false)
			humanoid.PlatformStand=true
			humanoid.AutoRotate = false
			momentum=torso.Velocity+(torso.CFrame.lookVector*10)+Vector3.new(0,3,0)

			local gyro=Instance.new("BodyGyro")
			gyro.Name="FlightGyro"
			gyro.P=10^6
			gyro.maxTorque=Vector3.new(gyro.P,gyro.P,gyro.P)
			gyro.cframe=torso.CFrame
			gyro.Parent=torso

			velocity=Instance.new("BodyVelocity")
			velocity.Name="FlightVelocity"
			velocity.velocity=Vector3.new(0,0,0)
			velocity.P=10^4
			velocity.maxForce=Vector3.new(1,1,1)*(10^6)
			velocity.Parent=torso
			
			local cloud=sp.Make:InvokeServer('Cloud')

			if lastanim~=nil then
				lastanim:Stop()
			end
			lastanim=humanoid:LoadAnimation(canim)
			lastanimused=0
			if lastanim then
				lastanim:Play(.5)
			end
			sp.Call:FireServer(wind, 'Play')
			while flying and torso and humanoid and humanoid.Health>0 and equipped do
				local cvector = Vector3.new(controls.left+controls.right,math.abs(controls.forward)*.2,controls.forward+controls.backward)
				local movement = Vector3.new()
				if cvector.magnitude > 0 then
					movement=game.Workspace.CurrentCamera.CoordinateFrame:vectorToWorldSpace(cvector.unit*speed)
				end

				momentum=(momentum*inertia)+movement
				totalmomentum=math.min(momentum.magnitude,topspeed)
				local momentumpercent=(totalmomentum/topspeed)
				if cloud~=nil then
					local m=cloud:FindFirstChild("Mesh")
					sp.Set:FireServer(m, 'Scale', Vector3.new(4,4,4+(momentumpercent*4)))
					m.Scale = Vector3.new(4,4,4+(momentumpercent*4))
				end
				sp.Set:FireServer(wind, 'Pitch', (momentumpercent*2)+1)
				sp.Set:FireServer(wind, 'Volume', momentumpercent+.1)

				local tilt=((momentum*Vector3.new(1,0,1)).unit:Cross(((lastmomentum*Vector3.new(1,0,1)).unit))).y
				if tostring(tilt)=="-1.#IND" or tostring(tilt)=="1.#IND" or tilt==math.huge or tilt==-math.huge or tostring(0/0) == tostring(tilt) then
					tilt=0
				end
				local abstilt=math.abs(tilt)
				if abstilt>.06 or abstilt<.0001 then
					if math.abs(lasttilt)>.0001 then
						tilt=lasttilt*.96
					else
						tilt=0
					end
				else
					tilt=(lasttilt*.9)+(tilt*.1)			--weighted average
				end
				lasttilt=tilt
				if tilt>.01 then
					if lastanimused~=1 then
						lastanim:Stop(.5)
						lastanim=humanoid:LoadAnimation(ranim)
						lastanim:Play(.5)
						lastanimused=1
					end
				elseif tilt<(-.01) then
					if lastanimused~=-1 then
						lastanim:Stop(.5)
						lastanim=humanoid:LoadAnimation(lanim)
						lastanim:Play(.5)
						lastanimused=-1
					end
				elseif lastanimused~=0 then
					lastanim:Stop(.5)
					lastanim=humanoid:LoadAnimation(canim)
					lastanim:Play(.5)
					lastanimused=0
				end
				if totalmomentum<.5 then
					momentum=Vector3.new(0,0,0)
					totalmomentum=0
					gyro.cframe=CFrame.new(Vector3.new(0,0,0),game.Workspace.CurrentCamera.CoordinateFrame.lookVector*Vector3.new(1,0,1))*CFrame.Angles(0,-math.pi/2,0)
				else
					gyro.cframe=CFrame.new(Vector3.new(0,0,0),momentum)*CFrame.Angles(0,-math.pi/2,0)*CFrame.Angles(tilt*(-20),0,0)
				end
				velocity.velocity=momentum
				lastmomentum=momentum
				game:GetService'RunService'.RenderStepped:wait()
			end
		end
		humanoid.AutoRotate = true
		RemoveFlyStuff()
		flying=false
	end
end

			local key=string.byte(key2)
			if key==string.byte("w") or key==17 then

			elseif key==string.byte("s") or key==18 then
				controls.backward=0
			elseif key==string.byte("d") or key==19 then
				controls.right=0
			end
		end)
		
		if Emulation then
			local keytodirection = {w="forward",a="left",s="backward",d="right"}
			local directionondown = {w=-1,a=-1,s=1,d=1}

	end

function createWeld(mainpart,target)
	local weld = Instance.new("Weld") 
	weld.Part0 = mainpart
	weld.Part1 = target 
	weld.C0 = mainpart.CFrame:inverse()

end

function RemoveFurnitureItem(plr,model) --referenced later down
	local house = workspace:FindFirstChild("!home_"..plr.Name)
	local stats = game.ReplicatedStorage.PlayerData:FindFirstChild(plr.userId)
	if house and stats and model.Parent == house then
		local itemserv = game.ReplicatedStorage.Objects:FindFirstChild(model.Name)
		if itemserv then
			local furnbought,pending = getFurnitureCostMarker(stats,itemserv)
			furnbought.Value = furnbought.Value + 1
			model:Destroy()
			if getOwnedObjectCount(plr,itemserv.ItemID.Value) + furnbought.Value <= 0 then
				stats.FurnitureCost.Value = stats.FurnitureCost.Value - itemserv.ItemCost.Value
			end
			getFurnitureCount(plr)
			return true
		end
	end
end

game.ReplicatedStorage.RemoteFunctions.GetFurniture.OnServerInvoke = function(plr,objname)
	local house = workspace:FindFirstChild("!home_"..plr.Name)
	local stats = game.ReplicatedStorage.PlayerData:FindFirstChild(plr.userId)
	if plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") and house and house.Floor.Editing.Value == true and stats then
		local torso = plr.Character.HumanoidRootPart
		local ray = Ray.new((torso.CFrame + torso.CFrame.lookVector * 10).p, (((torso.CFrame + torso.CFrame.lookVector * 10).p + Vector3.new(0,-15,0)) - (torso.CFrame + torso.CFrame.lookVector * 10).p).unit*15)
		local ignoredparts = {}
		for i,v in pairs(house:GetChildren()) do
			if v:IsA("Model") then
				table.insert(ignoredparts,v)
			end
		end
		--check if ray is inside house
		local findpart = workspace:FindPartOnRayWithIgnoreList(ray,ignoredparts)
		if findpart == nil or findpart.Parent.Name ~= house.Name or not findpart.Name:match("Floor") then
			return false,"Please stand facing some empty space inside your house, then try again."
		end
		local fcount = getFurnitureCount(plr)
		if fcount == 450 then
			game.ReplicatedStorage.RemoteFunctions.NotificationForClient:FireClient(plr, "Notice: You are approaching the item limit for your house. You have placed "..fcount.."/500 items.")
		end
		if fcount >= 500 then
			--hoarder!!
			return false,"Sorry, you cannot fit more than 1000 items in your home!"
		end
		
		
		local components = (torso.CFrame + torso.CFrame.lookVector * 10).p
		local itemserv = game.ReplicatedStorage.Objects:FindFirstChild(objname)
		if itemserv then
			local item = itemserv:clone()
			for _,v in pairs(item:GetChildren()) do
				if v.ClassName:match("Value") and v.Name ~= "PaintColor" and v.Name ~= "RotValue" and v.Name ~= "ItemID" then
					v:Destroy()
				end
			end
			item.Parent = house
			
			item:SetPrimaryPartCFrame(house.Floor.CFrame)
			local boxsize,boxcframe = require(game.ReplicatedStorage.Modules.GetBoundingBox)(item)
			local bottomboundary = boxcframe.Y - boxsize.Y / 2
			local yoffset = item.PrimaryPart.Position.Y - bottomboundary + house.Floor.Size.Y / 2
			local origpos = item.PrimaryPart.Position
			item:SetPrimaryPartCFrame(item.PrimaryPart.CFrame + (Vector3.new(math.floor(components.X),yoffset,math.floor(components.Z))) - Vector3.new(origpos.X,0,origpos.Z)) -- orig CFrame + (newpos - oldpos)

			local tempval = Instance.new("BoolValue")
			tempval.Name = "temp"
			tempval.Parent = item
			local furnbought,pending = getFurnitureCostMarker(stats,itemserv)
			furnbought.Value = furnbought.Value - 1
			if getOwnedObjectCount(plr,itemserv.ItemID.Value) + pending.Value + furnbought.Value < 0 then
				stats.FurnitureCost.Value = stats.FurnitureCost.Value + itemserv.ItemCost.Value
			end
			getFurnitureCount(plr)
			return true
		end
		
	end
end

game.ReplicatedStorage.RemoteFunctions.SellFurniture.OnServerInvoke = function(plr,objname)
	local stats = game.ReplicatedStorage.PlayerData:FindFirstChild(plr.userId)
	if stats then
		local servobj = game.ReplicatedStorage.Objects:FindFirstChild(objname)
		--local obj = stats.Objects:FindFirstChild(objname)
		if servobj and not servobj:FindFirstChild("TwitterCode") and not servobj:FindFirstChild("NotForSale") then
			local furnbought,pending = getFurnitureCostMarker(stats,servobj)
			if getOwnedObjectCount(plr,servobj.ItemID.Value) + pending.Value + furnbought.Value > 0 then
				pending.Value = pending.Value - 1
				stats.FurnitureCost.Value = stats.FurnitureCost.Value - math.ceil(servobj.ItemCost.Value * .7)
				return true
			end
		end
	end
end





game.ReplicatedStorage.RemoteFunctions.PaintHouseItem.OnServerEvent:connect(function(plr,item,color)
	if item.Parent.Name == "!home_"..plr.Name then
		if item:IsA("Model") then
			item.PaintColor.Value = color
			for i,v in pairs(item:GetChildren()) do
				if v:IsA("BasePart") and v.Name == "Paintable" then
					v.BrickColor = color
				end
			end
		elseif item:IsA("BasePart") then
			local data = game.ReplicatedStorage.PlayerData:FindFirstChild(plr.userId)
			
			local function paintParts(part,cl)
				for i,v in pairs(part.Parent:GetChildren()) do
					if v:IsA("BasePart") and v.Name == part.Name then
						v.BrickColor = cl
					end
				end
			end
			
			local colorbanned = false
			for _,v in pairs(bannedcolors) do
				if color.Name == v then
					colorbanned = true
				end
			end
			
			local isexterior = false
			for _,v in pairs(exteriorparts) do
				if item.Name == v then
					isexterior = true
				end
			end
			
			if not isexterior then
				paintParts(item,color)
			else
				if (data and (data.Objects:FindFirstChild("208") or game.ServerStorage.RemoteFunctions.GamePassCheck:Invoke(plr,"House"))) and colorbanned == false then
					paintParts(item,color)
				elseif colorbanned then
					paintParts(item,item.BrickColor)
					game.ReplicatedStorage.RemoteFunctions.NotificationForClient:FireClient(plr, "You're not allowed to paint your exterior walls with neon colors!")
				else
					paintParts(item,item.BrickColor)
					game.ReplicatedStorage.RemoteFunctions.OpenGuiFromServer:FireClient(plr,"BuyPaintingLicense")
				end
			end
			
			
		end
	end
end)



game.ReplicatedStorage.RemoteFunctions.SendNewFurniturePosition.OnServerEvent:connect(function(plr,selectedmodel,newcframe,newrotval)

		local pp = selectedmodel.PrimaryPart
		local ray = Ray.new(newcframe.p, ((newcframe.p + Vector3.new(0,-15,0)) - newcframe.p).unit*10)
		local ignoredparts = {}
		for i,v in pairs(selectedmodel.Parent:GetChildren()) do
			if v:IsA("Model") then
				table.insert(ignoredparts,v)
			end
		end
		local findpart = workspace:FindPartOnRayWithIgnoreList(ray,ignoredparts)
		if findpart == nil or not findpart.Name:match("Floor") then --put it back in orig pos for the client
			selectedmodel:SetPrimaryPartCFrame(pp.CFrame)

		end
	end

function PurchasePermanentItem(plr,itemid)
	local item
	local data = game.ReplicatedStorage.PlayerData:FindFirstChild(plr.userId)
	for i,v in pairs(game.ReplicatedStorage.Objects:GetChildren()) do
		if v.ItemID.Value == itemid then
			item = v
		end
	end
	local currenttime = game.ReplicatedStorage.IntroGUIData.CurrentTime.Value
	if item and item:FindFirstChild("NotForSale") == nil and not (item:FindFirstChild("ReleaseTime") and item.ReleaseTime.Value > currenttime and game.PlaceId == 9689581) and not (item:FindFirstChild("TimeToGoOffSale") and item.TimeToGoOffSale.Value < currenttime) and data and loadsuccess:Invoke(plr) == true then
		local plrobjects = data.Objects
		
		for i,v in pairs(plrobjects:GetChildren()) do
			if v.Name == tostring(itemid) then
				return false,"You already own this item! Please select another one."
			end
		end
		
		local cash = data.cash
		local price = item.ItemCost.Value
		if cash.Value >= price then
			coroutine.resume(coroutine.create(processPermanentItemPurchase), plr,price,itemid,item)
			if item.Name:match("gear_") then
				local cooldown = false
				for _,x in pairs(gearcooldown) do
					if x == plr.Name then
						cooldown = true
					end
				end
				if cooldown == false then
					coroutine.resume(coroutine.create(AddToCooldown),plr)	game.ReplicatedStorage.RemoteFunctions.NotificationForClient:FireClient(plr,"This Gear is now available via your RHS Inventory menu! (Press [G])")
				end
			end
			return true
		else
			return "cash",price - cash.Value
		end
	else
		game.ReplicatedStorage.RemoteFunctions.NotificationForClient:FireClient(plr,"ROBLOX's servers have encountered errors. To protect your data, you cannot make purchases during this session.")
	end

function Slerp(t, v1, v2)
	local om = math.acos(math.min(1, v1:Dot(v2)))
	return ((om < 0.01 and ((1 - t) * v1 + t * v2)) or ((math.sin((1 - t) * om) / math.sin(om)) * v1 + (math.sin(t * om) / math.sin(om)) * v2))
end

function TransformModel(Objects, Center, NewCFrame, Recurse)
	local Objects = ((type(Objects) ~= "table" and {Objects}) or Objects)
	for i, v in pairs(Objects) do
		if v:IsA("BasePart") then
			v.CFrame = NewCFrame:toWorldSpace(Center:toObjectSpace(v.CFrame))
		end
		if Recurse then
			TransformModel(v:GetChildren(), Center, NewCFrame, true)
		end
	end
end

function GetAllConnectedParts(Object)
	local Parts = {}
	local function GetConnectedParts(Object)
		for i, v in pairs(Object:GetConnectedParts()) do
			local Ignore = false
			for ii, vv in pairs(Parts) do
				if v == vv then
					Ignore = true
				end
			end
			if not Ignore then
				table.insert(Parts, v)
				GetConnectedParts(v)
			end
		end
	end
	GetConnectedParts(Object)
	return Parts
end

function ControlBalloon()
	
	if not CurrentBalloon or not CurrentBalloon.Balloon.Parent or not CurrentBalloon.MainPart.Parent or not CurrentBalloon.BodyGyro.Parent or not CurrentBalloon.BodyVelocity.Parent then
		return
	end
	
	local MainPart = CurrentBalloon.MainPart
	local FlyingForce = CurrentBalloon.BodyVelocity
	local FlyingGyro = CurrentBalloon.BodyGyro
	local FirePart1 = CurrentBalloon.FirePart1
	local Fire1 = CurrentBalloon.Fire1
	local Fire2 = CurrentBalloon.Fire2
	
	local SoundPlaying = false	
	
	local BalloonMass = 0
	for i, v in pairs(GetAllConnectedParts(MainPart)) do
		BalloonMass = (BalloonMass + v:GetMass())
	end
	BalloonMass = (BalloonMass * 2 * 196.20)
	FlyingForce.maxForce = Vector3.new(BalloonMass, BalloonMass, BalloonMass)
	
	local CurrentVelocity = math.min(MaxSpeed, FlyingForce.velocity.magnitude)	
	
	local LastTime = tick()
	local DesiredFlightCFrame = MainPart.CFrame
	
	while ToolEquipped and Flying do
		local Now = tick()
		local Time = (Now - LastTime)
		LastTime = Now
		
		if Flying then
			
			local CoordinateFrame = InvokeClient("GetCamera", {Property = "CoordinateFrame"})
			local MoveDirection = (CoordinateFrame * CFrame.Angles(0.6, 0, 0))
			if MouseDown then
				local Theta = math.acos(DesiredFlightCFrame.lookVector:Dot(MoveDirection.lookVector))
				local Frac = math.min(0.7, (0.7 * Time / Theta))
				local Unit = Slerp(Frac, DesiredFlightCFrame.lookVector, MoveDirection.lookVector)
				DesiredFlightCFrame = CFrame.new(Vector3.new(0, 0, 0), Unit)
				
				if CurrentVelocity < MaxSpeed then
					CurrentVelocity = math.min((CurrentVelocity + 8 * Time), MaxSpeed)
				end
			else
				if CurrentVelocity > 2 then
					CurrentVelocity = math.max(0, (CurrentVelocity - 3 * Time))
				end
			end
			
			local Direction = DesiredFlightCFrame.lookVector
			local DirectionXZ = Vector3.new(Direction.X, 0, Direction.Z).unit
			
			FlyingGyro.cframe = CFrame.new(Vector3.new(), DirectionXZ) * CFrame.Angles(0, 0, 0) --CFrame.Angles(0, 0, (MainPart.RotVelocity.y / 3 * (1 - math.abs(Direction.Y) ^ 2)))
			
			if FirePart1 and FirePart1.Parent then
				local FireSound = FirePart1:FindFirstChild("FireSound")
				if FireSound then
					if Direction.y > 0 then
						if not SoundPlaying then
							SoundPlaying = true
							FireSound:Play()
						end
					else
						SoundPlaying = false
						FireSound:Stop()
					end
				end
			end
			for i, v in pairs({Fire1, Fire2}) do
				if v and v.Parent then
					v.Enabled = MouseDown --(Direction.y > 0 and CurrentVelocity > 2)
				end
			end
			local SlowDownFactor = math.abs(MainPart.RotVelocity.Y / 3)
			FlyingForce.velocity = (Direction.unit * (CurrentVelocity - SlowDownFactor) * (1 - Direction.Y ^ 2))
		end
		wait()
	end
end

					Balloon.Parent = game:GetService("Workspace")
					SetUpConnections(CurrentBalloon)
				end
			else
				MouseDown = false
			end
		elseif mode == "KeyPress" then
			local Key = value.Key
			local ByteKey = string.byte(Key)
			local Down = value.Down
			if Key == "x" and Down then
				if CurrentBalloon and CurrentBalloon.Balloon.Parent then
					CurrentBalloon.Balloon:Destroy()
				end
			end
		end
	end
end

	end)

end
