function isOffScreen(point,bx,by) --bx and by are part of AbsoluteSize
	if point.x < 0 or point.x > bx or point.y < 0 or point.y > by then
		return true
	else
		return false
	end
end

function checkOffScreenCorners(element) --corners are like quadrants
	local pos = element.Position
	local size = element.Size
	local c1 = Vector2.new(pos.X.Offset + size.X.Offset, pos.Y.Offset)
	local c2 = Vector2.new(pos.X.Offset, pos.Y.Offset)
	local c3 = Vector2.new(pos.X.Offset, pos.Y.Offset + size.Y.Offset)
	local c4 = Vector2.new(pos.X.Offset + size.X.Offset, pos.Y.Offset + size.Y.Offset)
	local bx = container.AbsoluteSize.x
	local by = container.AbsoluteSize.y
	return isOffScreen(c1,bx,by),isOffScreen(c2,bx,by), isOffScreen(c3,bx,by),isOffScreen(c4,bx,by)
end

function MouseOverActiveFrame(x,y)
	local isover = false
	local function recurse(instance)
		for _,v in pairs(instance:GetChildren()) do
			if (v:IsA("Frame") or v:IsA("ScrollingFrame")) and v.Visible == true then
				if (x > v.AbsolutePosition.x and x < v.AbsolutePosition.x + v.AbsoluteSize.x and y > v.AbsolutePosition.y and y < v.AbsolutePosition.y + v.AbsoluteSize.y) then
					isover = true
					break
				else
					recurse(v)
				end
			end
		end
	end
	if gui:FindFirstChild("MainMenu") then
		recurse(gui.MainMenu)
	end
	if not isover then
		if gui:FindFirstChild("PhoneGUI") and gui.PhoneGUI.Frame.Visible == true then
			recurse(gui.PhoneGUI)
		end
	else
		return true
	end
	if not isover then
		if gui:FindFirstChild("HouseEditMode") and gui.HouseEditMode.CenterFrame.Visible == true then
			recurse(gui.HouseEditMode.CenterFrame)
		end
	else
		return true
	end
	if not isover then
		return false
	else
		return true
	end
end

function RepeatGet(pd, key, successes, tries)
	local success, reason, returnvalue
	local i = 0
	repeat
		if i ~= 0 then
			wait(3)
		end
		success, reason = pcall(function()
			returnvalue = pd:GetAsync(key)
		end)
		i = i + 1
	until success or i >= (tries or 10)
	
	if successes then
		table.insert(successes,success)
		if debugmode or not success then print(reason) end
	end
	
	return returnvalue, success, reason
end

	local function GetLoyaltyData(tbl)
		local changes = false
		local timedata = game.ReplicatedStorage.IntroGUIData
		local day = timedata.CurrentRealLifeDay.Value
		local endtime = timedata.HolidayStartTime.Value + (day * 86400)
		local timeleft = endtime - timedata.CurrentTime.Value
		local timeleftstring = math.floor(timeleft/3600).."h "..math.floor((timeleft%3600)/60).."m"
		
		local current_lp = tbl[1]
		local lp_reward = 0
		local consec_days = tbl[2]
		local current_spins = tbl[4]
		local firstlogintoday = false
		local lastdayloggedin = tbl[3]
		
		if tbl[3] ~= day then --hasn't logged in today
			firstlogintoday = true
			changes = true
			if day - tbl[3] <= 2 then --logged in yesterday or 2 days before, offer rewards for consec. logins
				tbl[3] = day
				tbl[2] = tbl[2] + 1
				consec_days = tbl[2]
				if tbl[2] >= 5 then
					lp_reward = 1000
				else
					lp_reward = 500 + tbl[2] * 100
				end
				tbl[1] = tbl[1] + lp_reward
				current_lp = tbl[1]
				if consec_days % 5 == 0 then
					tbl[4] = tbl[4] + 2
				end
				current_spins = tbl[4]
			else --didn't log in yesterday, reset counter
				tbl[2] = 1
				consec_days = 1
				lp_reward = 600
				tbl[1] = tbl[1] + lp_reward
				current_lp = tbl[1]
				tbl[3] = day
				current_spins = tbl[4]
			end
		end
		
		--create stats
		local loyaltystats = Instance.new("Folder")
		loyaltystats.Name = "LoyaltyStats"
		
		local lp = Instance.new("IntValue",loyaltystats)
		lp.Name = "LP"
		lp.Value = current_lp
		local oldlp = Instance.new("IntValue",loyaltystats)
		oldlp.Name = "OldLP"
		oldlp.Value = current_lp
		
		local spins = Instance.new("IntValue",loyaltystats)
		spins.Name = "Spins"
		spins.Value = current_spins
		local oldspins = Instance.new("IntValue",loyaltystats)
		oldspins.Name = "OldSpins"
		oldspins.Value = current_spins
		
		local lastday = Instance.new("IntValue",loyaltystats)
		lastday.Name = "LastDayLoggedIn"
		lastday.Value = lastdayloggedin
		
		--store data that is only used to determine what's displayed on GUI
		local rewardstat = Instance.new("IntValue",loyaltystats)
		rewardstat.Name = "LPReward"
		rewardstat.Value = lp_reward
		local daysstat = Instance.new("IntValue",loyaltystats)

		local playerloginday = Instance.new("IntValue",loyaltystats)
		playerloginday.Name = "PlayerLoginDay"
		playerloginday.Value = day
		
		loyaltystats.Parent = stats	
		
		return tbl, changes
	end
