local gui = script.Parent:WaitForChild("Frame")
local plr = game.Players.LocalPlayer
local enabled = gui:WaitForChild("Enabled")

	savedcategory = Instance.new("StringValue",plr)

	savedcategory.Value = "Seating"

local data = game.ReplicatedStorage.PlayerData:WaitForChild(plr.userId)

function getAvailableItemCount(id)
	local count = 0
	for _,v in pairs(data.Objects:GetChildren()) do
		if tonumber(v.Name) == id then
			count = count + 1
		end
	end
	local furnbought = data.FurnitureCost:FindFirstChild(id)
	if furnbought then
		count = count + furnbought.PendingSales.Value + furnbought.Value --items in Objects - sold items - added items (or + removed items)
	end
	return count
end

--now load all the buttons
function loadCategory(categoryname)
	if enabled.Value == true then
		local xv = 0
		local yv = 0
		
		selectedcategory.Value = categoryname
		savedcategory.Value = categoryname
		
		for _,v in pairs(gui.Items:GetChildren()) do
			if v.Name ~= "BaseButton" then
				v:Destroy()
			end
		end
		
		for i,v in pairs(shop) do
			if v.Category.Value == selectedcategory.Value and not (v:FindFirstChild("NotForSale") and getAvailableItemCount(v.ItemID.Value) <= 0) and not (v:FindFirstChild("ReleaseTime") and v.ReleaseTime.Value > game.ReplicatedStorage.IntroGUIData.CurrentTime.Value) then
				local button = gui.Items.BaseButton:clone() --make new button for each item
				button.Parent = gui.Items
				button.Name = v.Name
				button.ItemID.Value = v.ItemID.Value
				button.Image = "http://www.roblox.com/Thumbs/Asset.ashx?width=75&height=75&assetId=".. v.ModelID.Value
				button.Position = UDim2.new(0, xv*75, 0, yv*75)
				if v:FindFirstChild("TwitterCode") then
					local bird = script.TwitterBird:clone()
					bird.Parent = button
					bird.Visible = true
				end
				if v:FindFirstChild("ReleaseTime") and game.ReplicatedStorage.IntroGUIData.CurrentTime.Value - v.ReleaseTime.Value < 86400*21 then
					local newb = script.NewBanner:clone()
					newb.Parent = button
					newb.Visible = true
				end
				xv = xv + 1
				if xv == 4 then
					xv = 0
					yv = yv + 1
				end
				button.Visible = true
			end
		end
		
		gui.Items.CanvasSize = UDim2.new(0,0,0,(yv + 1) * 75)
		gui.LoadItemInfo:Fire()
		setButtonColors()
		
		for _,v in pairs(gui.Nav:GetChildren()) do
			if v.Name == categoryname then
				v.BackgroundTransparency = 0
			else
				v.BackgroundTransparency = .5
			end
		end
	end
end

local module = require(game.ReplicatedStorage.Modules.InterfaceModule)
module.openWindow(gui)


function setButtonColors()
	for _,button in pairs(gui.Items:GetChildren()) do
		if button.Name ~= "BaseButton" then
			if getAvailableItemCount(button.ItemID.Value) > 0 then --has object
				button.BackgroundColor3 = Color3.new(.5,1,.5)
			else
				button.BackgroundColor3 = Color3.new(1,1,1)
			end
			if gui.SelectedItem.Value == button.ItemID.Value then
				button.BackgroundColor3 = Color3.new(1,1,.25)
			end
		end
	end
end


function displayFailureWindow(result,reason) --reason == amount of cash needed
	if result == false then
		
		local failure = gui.FailureWindow
		failure.Header.Text = "Oops!"
		failure.Reason.Text = reason
		failure.NormalButton.Visible = true
		failure.OptionButtons.Visible = false
		failure.Visible = true
		repeat wait() until failure.Visible == false
		
	elseif result == "sell" then --reason is the obj
		local failure = gui.FailureWindow
		failure.Header.Text = "Sell Item? (70%)"
		failure.Reason.Text = "Sell this item for $"..math.ceil(reason.ItemCost.Value * .7).."? This will be deducted from your final expenses."
		failure.NormalButton.Visible = false
		failure.OptionButtons.Visible = true
		failure.Visible = true
		repeat wait() until failure.Visible == false
		
		for _,v in pairs(gui.Items:GetChildren()) do
			if v.ItemID.Value == gui.SelectedItem.Value then
				gui.LoadItemInfo:Fire(v)
				break
			end
		end
	end
end


gui.AddFurniture.Event:connect(function(button)
	if enabled.Value == true then
		local obj
		for _,v in pairs(game.ReplicatedStorage.Objects:GetChildren()) do
			if v.Name:match("furni_") and v.ItemID.Value == gui.SelectedItem.Value then
				obj = v
			end
		end
		if obj then
			enabled.Value = false
			local oldbg = button.BackgroundColor3
			local oldborder = button.BorderColor3
			button.BackgroundColor3 = Color3.new(.6,.6,.6)
			button.BorderColor3 = Color3.new(.1,.1,.1)
			local result,reason = game.ReplicatedStorage.RemoteFunctions.GetFurniture:InvokeServer(obj.Name)
			if result == true then
				module.closeWindow(script.Parent)
			elseif result == false then
				displayFailureWindow(result,reason)
			end
			button.BackgroundColor3 = oldbg
			button.BorderColor3 = oldborder
			enabled.Value = true
		end
	end
end)

gui.SellFurniture.Event:connect(function()
	if enabled.Value == true then
		local obj
		for _,v in pairs(game.ReplicatedStorage.Objects:GetChildren()) do
			if v.Name:match("furni_") and v.ItemID.Value == gui.SelectedItem.Value then
				obj = v
			end
		end
		if obj then
			enabled.Value = false
			displayFailureWindow("sell",obj)

	end
