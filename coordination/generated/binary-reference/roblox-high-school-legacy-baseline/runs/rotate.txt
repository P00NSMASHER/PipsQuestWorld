function setFeedback(type,text,value) --types: Save, Exit, Dialog (no buttons), NormalButton, Cash

	gui.WindowType.Value = type
	if text == nil then
		text = ""
	end
	gui.Reason.Text = text
	for _,v in pairs(gui:GetChildren()) do
		if v:IsA("Frame") then
			v.Visible = false
		end
	end
	if type == "Save" or type == "Exit" then
		gui.OptionButtons.Visible = true
	--elseif type == "Dialog" then	
	elseif type == "NormalButton" then
		gui[type].Visible = true
	elseif type == "Cash" then

		local cash = data.cash.Value
		local products = game.ReplicatedStorage.CashProducts:GetChildren()
		local bestprod
		local highestprod
		for i,v in pairs(products) do
			if v.Name ~= "30" then
				local diff = tonumber(v.Name) - value
				if diff >= 0 and (bestprod == nil or tonumber(bestprod.Name) > tonumber(v.Name)) then
					bestprod = v
				end
				if highestprod == nil or tonumber(v.Name) > tonumber(highestprod.Name) then
					highestprod = v
				end
			end
		end
		if bestprod == nil then
			bestprod = highestprod
		end
		
		gui.Reason.Text = "You need more Cash in order to save all the items that you've added to your house. Get more Cash & save the house?"
		gui.CashAmount.Value = tonumber(bestprod.Name)
		gui.ProductId.Value = bestprod.Value
		gui.BuyCashButtons.GetCash.Text = "Get +"..bestprod.Name.." Cash!"
		gui.BuyCashButtons.Visible = true
		
	end
	gui.Visible = true

function CheckTableForString(Table, String)
	for i, v in pairs(Table) do
		if string.find(string.lower(String), string.lower(v)) then
			return true
		end
	end
	return false

function CheckIntangible(Hit)
	local ProjectileNames = {"Water", "Arrow", "Projectile", "Effect", "Rail", "Laser", "Bullet"}
	if Hit and Hit.Parent then
		if ((not Hit.CanCollide or CheckTableForString(ProjectileNames, Hit.Name)) and not Hit.Parent:FindFirstChild("Humanoid")) then
			return true
		end
	end
	return false
end

function CastRay(StartPos, Vec, Length, Ignore, DelayIfHit)
	local Ignore = ((type(Ignore) == "table" and Ignore) or {Ignore})
	local RayHit, RayPos, RayNormal = game:GetService("Workspace"):FindPartOnRayWithIgnoreList(Ray.new(StartPos, Vec * Length), Ignore)
	if RayHit and CheckIntangible(RayHit) then
		if DelayIfHit then
			wait()
		end
		RayHit, RayPos, RayNormal = CastRay((RayPos + (Vec * 0.01)), Vec, (Length - ((StartPos - RayPos).magnitude)), Ignore, DelayIfHit)
	end
	return RayHit, RayPos, RayNormal
end

function GetAllConnectedParts(Object)
	local Parts = {}
	local function GetConnectedParts(Object)
		for i, v in pairs(Object:GetConnectedParts()) do
			local Ignore = false
			for ii, vv in pairs(Parts) do
				if v == vv then
					Ignore = true
				end
			end
			if not Ignore then
				table.insert(Parts, v)
				GetConnectedParts(v)
			end
		end
	end
	GetConnectedParts(Object)
	return Parts
end

function SetGravityEffect(Table)
	local Table = (Table or {})
	local Multiplier = (Table.Multiplier or 1)
	if not GravityEffect or not GravityEffect.Parent then
		GravityEffect = Create(Table.Class or "BodyForce"){
			Name = "GravityCoilEffect",
		}
		for i, v in pairs(Table.Properties or {}) do
			pcall(function()
				GravityEffect[i] = v
			end)
		end
	end
	local TotalMass = 0
	local ConnectedParts = GetAllConnectedParts(Torso)
	for i, v in pairs(ConnectedParts) do
		if v:IsA("BasePart") then
			TotalMass = (TotalMass + v:GetMass())
		end
	end
	local TotalMass = ((TotalMass * Gravity) * Multiplier)
	if GravityEffect:IsA("BodyVelocity") then
		GravityEffect.maxForce = Vector3.new(0, TotalMass, 0)
	elseif GravityEffect:IsA("BodyForce") then
		GravityEffect.force = Vector3.new(0, TotalMass, 0)
	end
	GravityEffect.Parent = Torso
	return GravityEffect
end

    return res
  end
  return bitop
end

local function make_bitop(t)
  local op1 = make_bitop_uncached(t,2^1)
  local op2 = memoize(function(a)
    return memoize(function(b)
      return op1(a, b)
    end)
  end)
  return make_bitop_uncached(op2, 2^(t.n or 1))
end
