	if Wire then

	Wire = Instance.new('Part',Bin)
	Wire.CanCollide = false
	Wire.Anchored = true
	Wire.BrickColor = BrickColor.new('Light stone grey')

	local Mesh = Instance.new('CylinderMesh', Wire)
	Mesh.Scale = Vector3.new(1, 1, 0.3)

function throw()
	if Mouse.Hit and Mouse.Target and Mouse.Target.Name == "Water" then
		DrawFromPoints(WireStart.Position, Mouse.Hit.p)

function vowelCheck(fish)
	local vowels = {"a","e","i","o","u"}
	for i,v in pairs(vowels) do
		if string.sub(string.lower(fish),1,1) == v then
			return true
		end
	end
end

function isOffScreen(point,bx,by) --bx and by are part of AbsoluteSize
	if point.x < 0 or point.x > bx or point.y < 0 or point.y > by then
		return true

function checkOffScreenCorners(element) --corners are like quadrants
	local pos = element.Position
	local size = element.Size
	local c1 = Vector2.new(pos.X.Offset + size.X.Offset, pos.Y.Offset)
	local c2 = Vector2.new(pos.X.Offset, pos.Y.Offset)
	local c3 = Vector2.new(pos.X.Offset, pos.Y.Offset + size.Y.Offset)
	local c4 = Vector2.new(pos.X.Offset + size.X.Offset, pos.Y.Offset + size.Y.Offset)

function MouseOverActiveFrame(x,y)
	local isover = false
	local function recurse(instance)
		for _,v in pairs(instance:GetChildren()) do
			if (v:IsA("Frame") or v:IsA("ScrollingFrame")) and v.Visible == true then
				if (x > v.AbsolutePosition.x and x < v.AbsolutePosition.x + v.AbsoluteSize.x and y > v.AbsolutePosition.y and y < v.AbsolutePosition.y + v.AbsoluteSize.y) then
					isover = true
					break
				else
					recurse(v)
				end
			end
		end
	end
	if gui:FindFirstChild("MainMenu") then
		recurse(gui.MainMenu)
	end
	if not isover then
		if gui:FindFirstChild("PhoneGUI") and gui.PhoneGUI.Frame.Visible == true then
			recurse(gui.PhoneGUI)
		end
	else
		return true
	end
	if not isover then
		if gui:FindFirstChild("HouseEditMode") and gui.HouseEditMode.CenterFrame.Visible == true then
			recurse(gui.HouseEditMode.CenterFrame)
		end
	else
		return true
	end
	if not isover then
		return false
	else
		return true
	end
end

function TextTransparencyFade(element, totalframes, fadein)
	coroutine.resume(coroutine.create(function()
		local i = 0
		local transincrement = (1 - element.TextTransparency) / totalframes
		local strokeincrement = (1 - element.TextStrokeTransparency) / totalframes
		local mult = 1
		if fadein then
			mult = -1
			element.TextTransparency = 1
			element.TextStrokeTransparency = 1
		end
		element.Visible = true
		local render = rs.RenderStepped:connect(function()
			if i < totalframes then
				element.TextTransparency = element.TextTransparency + transincrement * mult
				element.TextStrokeTransparency = element.TextStrokeTransparency + strokeincrement * mult
				i = i + 1
			end
		end)
		repeat wait() until i >= totalframes
		if fadein == false then
			element.Visible = false
			element.TextTransparency = 1 - (transincrement * totalframes)
			element.TextStrokeTransparency = 1 - (strokeincrement * totalframes)
		end
		render:disconnect()
	end),element, totalframes, fadein)
end

function ImageFade(element, totalframes, fadein)
	coroutine.resume(coroutine.create(function()
		local i = 0
		local increment = 1 / totalframes
		local mult = 1
		if fadein then
			mult = -1
			element.ImageTransparency = 1
		else
			element.ImageTransparency = 0
		end
		element.Visible = true
		local render = rs.RenderStepped:connect(function()
			if i < totalframes then
				element.ImageTransparency = element.ImageTransparency + increment * mult
				i = i + 1
			end
		end)
		repeat wait() until i >= totalframes
		if fadein == false then
			element.Visible = false
		end
		render:disconnect()
	end),element, totalframes, fadein)
end

function ShiftVolume(sound, starting)

	shifting = true

	elseif highest_loudness <= 400 then

	elseif highest_loudness <= 500 then
		target_volume = .5
	elseif highest_loudness <= 675 then
		target_volume = .15 + (675 - highest_loudness) / 500
	elseif highest_loudness > 675 then
		target_volume = .1
	end
	
	if starting ~= true and sound.Volume <= target_volume then shifting = false return end
	
	local vol = sound.Volume
	for i=1,7 do
		if sound then
			sound.Volume = sound.Volume - ((vol - target_volume) / 7)

			end

function getBackgroundMusic() --get BG music based on time/day
	local day = game.ReplicatedStorage.Day.Value
	local wp = weekdayspassed.Value
	
	local t = game.Lighting:GetMinutesAfterMidnight()
	if t >= 1380 then --get next day's music (late night)
		day = day + 1
		if day == 8 then
			day = 1
		end
	end
	
	local musicset
	if day == 1 or day == 7 then
		musicset = 4
	else
		musicset = (weekdayspassed.Value - 1) % 3 + 1
	end
	
	local segment
	if t >= 1380 or t < 300 then
		segment = 1
	elseif t >= 300 and t < 660 then
		segment = 2
	elseif t >= 660 and t < 1050 then
		segment = 3
	elseif t >= 1050 and t < 1380 then
		segment = 4
	end
	
	local id = "rbxassetid://"..songs[musicset][segment]
	return id,musicset,segment
end

function addToHandling(plr)
	table.insert(handling,plr.Name)

function canHandlePlayer(plr)
	local count = 0
	for i,v in pairs(handling) do
		if v == plr.Name then
			count = count + 1
		end
	end
	if count > 1 then return false end
	coroutine.resume(coroutine.create(addToHandling),plr)
	return true
end

function GetDataStoreSafeString(str)
	if not str then return end
	for i=1,#str do
		local letter = string.sub(str,i,i)
		if string.byte(letter) > 127 then
			return str:match("%w+") or "Invalid Name"
		end
	end
	return str
end

function GetFilteredNames(player, outfitname, rpname, rpdesc)
	local flag
	pcall(function()
		flag = game:GetService("Chat"):CanUserChatAsync(player.userId)
	end)
	if not flag then
		outfitname = "Saved Outfit"
		rpname = ""
		rpdesc = ""
		return outfitname, rpname, rpdesc
	end
	outfitname = GetDataStoreSafeString(outfitname)
	rpname = GetDataStoreSafeString(rpname)
	rpdesc = GetDataStoreSafeString(rpdesc)
	
	if outfitname and outfitname ~= "" then
		local filteredname
		pcall(function()
			filteredname = game:GetService("Chat"):FilterStringAsync(outfitname, player, player)
		end)
		if filteredname then
			if filteredname ~= outfitname then
				outfitname = "Saved Outfit"
			end
		end
	end
	
	if rpname and rpname ~= "" then
		for i=1,#rpname do
			local letter = string.sub(rpname,i,i)
			if letter:match("%A") and letter ~= " " and letter ~= "." then
				rpname = "INVALID_NAME"
			end
		end
		
		if not rpname:match("%a") then
			rpname = "INVALID_NAME"
		end
		
		if rpname ~= "INVALID_NAME" then
			local filteredname
			pcall(function()
				filteredname = game:GetService("Chat"):FilterStringAsync(rpname, player, player)
			end)
			
			if filteredname then
				if filteredname ~= rpname then
					if player then
						rpname = player.Name:match("%a+")
					end
					if rpname == nil then
						rpname = "Charlie"
					end
					rpname = string.sub(rpname,1,15)
				end
			end
		end
	end
	
	if rpdesc and rpdesc ~= "" then
		local pattern = [[

]]
		
		if rpdesc:match(pattern) then
			rpdesc = "INVALID_NAME"
		end
		
		if rpdesc ~= "INVALID_NAME" then
			local filteredname
			pcall(function()
				filteredname = game:GetService("Chat"):FilterStringAsync(rpdesc, player, player)
			end)
			if filteredname then
				rpdesc = filteredname
			end
		end
	end
	
	return outfitname, rpname, rpdesc
end

function checkOutfit(plr,inputs,wearonly) --don't check outfit name if wearonly == true
	local statuses = {}
	for i,v in pairs(inputs) do
		statuses[i] = "invalid" --give us a fresh table of the input names w/ statuses
	end
	
	inputs["OutfitName"], inputs["RPName"], inputs["RPDesc"] = GetFilteredNames(plr, inputs["OutfitName"], inputs["RPName"], inputs["RPDesc"])
	
	for i,v in pairs(inputs) do
		if i == "OutfitName" then
			if v == nil or ((v == "" or #v > 25) and not wearonly) or v == "INVALID_NAME" then
				statuses[i] = "invalid"
			else
				statuses[i] = 0
			end
		elseif i == "RPName" then
			if v == nil or (inputs["RPDesc"] ~= "" and inputs["RPName"] == "") or v == "INVALID_NAME" then
				statuses["RPName"] = "invalid"
			else
				statuses[i] = 0
			end
		elseif i == "RPDesc" then
			if v == nil or v == "INVALID_NAME" then
				statuses[i] = "invalid"
			else
				statuses[i] = 0
			end
		elseif i == "RemoveShirt" then
			statuses[i] = 0
		elseif v == "" then
			statuses[i] = "noload"
		elseif tostring(v):match("%D") then
			statuses[i] = "invalid"
		else --gettin' to the beefy stuff
			local asset = getProductInfo(v)
			if asset then
				--local creator = asset.Creator["Name"] --the creator's username
				
				if i:match("Hat") then
					if asset.AssetTypeId == 8 or (asset.AssetTypeId >= 41 and asset.AssetTypeId <= 47) then
						if (asset.AssetId ~= 72082328 or plr.Name == "Cindering") and (asset.AssetId ~= 62679779 or plr.Name ~= "Ultimateguy") and asset.AssetId ~= 1527622 and asset.AssetId ~= 1055299 then
							statuses[i] = 0
						elseif asset.AssetId == 501960266 and not (plr:FindFirstChild("HasRHSToy") and plr.HasRHSToy.Value == true) then --Has RHS Headphones
							statuses[i] = "invalid"
							game.ReplicatedStorage.RemoteFunctions.NotificationForClient:FireClient(plr, "Earn the exclusive RHS Headphones item by purchasing a Roblox High School physical toy pack! Visit roblox.com/toys to learn more.", true)
						else
							statuses[i] = "admin"
						end
					else
						statuses[i] = "invalid"
					end
				elseif i == "Shirt" or i == "Pants" then
					if asset.AssetTypeId == 1 then --it's already an Image
						statuses[i] = 0
					elseif asset.AssetTypeId == 11 or asset.AssetTypeId == 12 then
						local count = 0
						local imgasset
						local success
						repeat
							pcall(function() imgasset = mp:GetProductInfo(v-(count+1)) end)
							if imgasset and imgasset.AssetTypeId == 1 then
								success = true
							else
								imgasset = nil
							end
							count = count + 1
						until success == true or count > 4
						if success == true then
							statuses[i] = count
						else
							statuses[i] = "invalid"
						end
					else
						statuses[i] = "invalid"
					end
				elseif i == "Face" then
					if asset.AssetTypeId == 18 then
						local textureid
						local faceasset
						pcall(function() faceasset = game:GetService("InsertService"):LoadAsset(v) textureid = (faceasset:GetChildren()[1].Texture):match("%d+") end)
						if faceasset then
							faceasset:Destroy()
						end
						if textureid then
							statuses[i] = v - tonumber(textureid)
						end
					end
				end
				
			else
				statuses[i] = "invalid"
			end
			
		end
	end
	
	return statuses
	
end
