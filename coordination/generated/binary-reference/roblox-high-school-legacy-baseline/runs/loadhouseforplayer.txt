function createWeld(mainpart,target)
	local weld = Instance.new("Weld") 
	weld.Part0 = mainpart
	weld.Part1 = target 
	weld.C0 = mainpart.CFrame:inverse() 
	weld.C1 = target.CFrame:inverse() 
	weld.Parent = target
	target.Anchored = false
end

function getFurnitureCount(plr)
	local data = game.ReplicatedStorage.PlayerData:FindFirstChild(plr.userId)
	local house = workspace:FindFirstChild("!home_"..plr.Name)
	if data and house then
		local val = data:FindFirstChild("FurnitureCount")
		if val == nil then
			val = Instance.new("IntValue",data)
			val.Name = "FurnitureCount"
		end
		local fcount = 0
		for i,v in pairs(house:GetChildren()) do
			if v:IsA("Model") then
				fcount = fcount + 1
			end
		end
		val.Value = fcount
		return fcount
	end
end

function getFurnitureCostMarker(stats,servobj)
	local furnbought
	local id = servobj.ItemID.Value
	if stats.FurnitureCost:FindFirstChild(id) then
		furnbought = stats.FurnitureCost:FindFirstChild(id)
	else
		furnbought = Instance.new("IntValue")
		furnbought.Value = 0
		furnbought.Name = id
		furnbought.Parent = stats.FurnitureCost
	end
	local pending
	if furnbought:FindFirstChild("PendingSales") then
		pending = furnbought.PendingSales
	else
		pending = Instance.new("IntValue")
		pending.Value = 0
		pending.Name = "PendingSales"
		pending.Parent = furnbought
	end
	return furnbought,pending
end

function RemoveFurnitureItem(plr,model) --referenced later down
	local house = workspace:FindFirstChild("!home_"..plr.Name)
	local stats = game.ReplicatedStorage.PlayerData:FindFirstChild(plr.userId)
	if house and stats and model.Parent == house then
		local itemserv = game.ReplicatedStorage.Objects:FindFirstChild(model.Name)
		if itemserv then
			local furnbought,pending = getFurnitureCostMarker(stats,itemserv)
			furnbought.Value = furnbought.Value + 1
			model:Destroy()
			if getOwnedObjectCount(plr,itemserv.ItemID.Value) + furnbought.Value <= 0 then
				stats.FurnitureCost.Value = stats.FurnitureCost.Value - itemserv.ItemCost.Value
			end
			getFurnitureCount(plr)
			return true
		end
	end
end

function RepeatGet(pd, key, successes, tries)
	local success, reason, returnvalue
	local i = 0
	repeat
		if i ~= 0 then
			wait(3)
		end
		success, reason = pcall(function()
			returnvalue = pd:GetAsync(key)
		end)
		i = i + 1
	until success or i >= (tries or 10)
	
	if successes then
		table.insert(successes,success)
		if debugmode or not success then print(reason) end
	end
	
	return returnvalue, success, reason
end

function processMultipleAwardItems(plr,IDtable)
	local data = game.ReplicatedStorage.PlayerData:FindFirstChild(plr.userId)
	if data and loadsuccess:Invoke(plr) == true then
		local success = pcall(function()
			local update = dss:GetDataStore(plr.userId.."_data"):UpdateAsync("maindata",function(tbl)
				if tbl == nil then
					return tbl
				end
				for _,itemid in pairs(IDtable) do
					table.insert(tbl["objects"],itemid)
				end
				return tbl
			end)
		end)
		if success then
			for _,itemid in pairs(IDtable) do
				local marker = Instance.new("StringValue",data.Objects)
				marker.Name = itemid
			end
			return true
		else
			game.ReplicatedStorage.RemoteFunctions.NotificationForClient:FireClient(plr,"An error occurred while awarding items. Please try again.")
		end
	else
		game.ReplicatedStorage.RemoteFunctions.NotificationForClient:FireClient(plr,"ROBLOX's servers have encountered errors. Your data will not be modified during this session.")
	end
end

game.ReplicatedStorage.RemoteFunctions.RedeemTwitterCode.OnServerInvoke = function(plr,code)
	local data = game.ReplicatedStorage.PlayerData:FindFirstChild(plr.userId)
	if data then
		
		local obj 
		for _,v in pairs(game.ReplicatedStorage.Objects:GetChildren()) do
			if v:FindFirstChild("TwitterCode") and string.lower(v.TwitterCode.Value) == string.lower(code) then --code correct
				obj = v
			end
		end
		if obj == nil then
			return false,"Error: You have entered an incorrect code. Please enter a valid code that was given away in the RHS Fan Club."
		end
		
		local itemname = string.sub(obj.Name:match("_.+"),2)
		
		for _,v in pairs(data.Objects:GetChildren()) do
			if tonumber(v.Name) == obj.ItemID.Value then
				return false,"You've already redeemed this code for '"..itemname.."'!"
			end
		end
		
		coroutine.resume(coroutine.create(processAwardItem),plr,obj.ItemID.Value)
		return true,"Successfully redeemed code for '"..itemname.."'!"
		
	end
end
