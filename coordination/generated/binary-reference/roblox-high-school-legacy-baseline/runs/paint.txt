	if timeleft < 0 then
		return "Ends now!"

			else

			end

			else

function ApplyPaint(plr, paintmodel, canvas, position, mode, brickcolor)
	local part = Instance.new("Part")
	part.Name = "ServerPart"
	if game:GetService("RunService"):IsClient() then
		part.Name = "LocalPart"
	end
	part.Shape = "Ball"
	part.Size = Vector3.new(.3,.3,.3)
	part.CFrame = CFrame.new(position)
	part.Anchored = true
	
	if mode == "Paint" then
		part.BrickColor = brickcolor or BrickColor.Red()
		part.TopSurface = "Smooth"
		part.BottomSurface = "Smooth"
		part.Material = "SmoothPlastic"
	else
		part.Transparency = 1
	end
	
	if mode == "Erase" then
		for _,v in pairs(paintmodel:GetChildren()) do
			if v:IsA("Part") then
				v.CanCollide = true
			end
		end
	end
	
	part.Parent = paintmodel
	
	local touchingparts = part:GetTouchingParts()
	
	part.Parent = paintmodel
	
	if mode == "Erase" then
		for _,v in pairs(paintmodel:GetChildren()) do
			if v:IsA("Part") then
				v.CanCollide = false
			end
		end
	end
	
	part.CanCollide = false
	
	if mode == "Paint" then
		local istouching = false
		for _,v in pairs(touchingparts) do

			end
		end

		if game:GetService("RunService"):IsClient() then
			local request = game.ReplicatedStorage.RemoteFunctions.PlacePaintRequest:InvokeServer(canvas, position, mode, brickcolor)
			if part then
				part:Destroy()
			end
		end
		
		return true
	elseif mode == "Erase" then
		for _,v in pairs(touchingparts) do
			if v.Parent == paintmodel then
				v:Destroy()
			end
		end
		
		part:Destroy()

			canvas.CanvasOwner.Value = nil
		end
		
		if game:GetService("RunService"):IsClient() then
			local request = game.ReplicatedStorage.RemoteFunctions.PlacePaintRequest:InvokeServer(canvas, position, mode)
		end

function RepeatGet(pd, key, successes, tries)
	local success, reason, returnvalue
	local i = 0
	repeat
		if i ~= 0 then
			wait(3)
		end
		success, reason = pcall(function()
			returnvalue = pd:GetAsync(key)
		end)
		i = i + 1
	until success or i >= (tries or 10)
	
	if successes then
		table.insert(successes,success)
		if debugmode or not success then print(reason) end
	end
	
	return returnvalue, success, reason
end

	local function GetLoyaltyData(tbl)
		local changes = false
		local timedata = game.ReplicatedStorage.IntroGUIData
		local day = timedata.CurrentRealLifeDay.Value
		local endtime = timedata.HolidayStartTime.Value + (day * 86400)
		local timeleft = endtime - timedata.CurrentTime.Value
		local timeleftstring = math.floor(timeleft/3600).."h "..math.floor((timeleft%3600)/60).."m"
		
		local current_lp = tbl[1]
		local lp_reward = 0
		local consec_days = tbl[2]
		local current_spins = tbl[4]
		local firstlogintoday = false
		local lastdayloggedin = tbl[3]
		
		if tbl[3] ~= day then --hasn't logged in today
			firstlogintoday = true
			changes = true
			if day - tbl[3] <= 2 then --logged in yesterday or 2 days before, offer rewards for consec. logins
				tbl[3] = day
				tbl[2] = tbl[2] + 1
				consec_days = tbl[2]
				if tbl[2] >= 5 then
					lp_reward = 1000
				else
					lp_reward = 500 + tbl[2] * 100
				end
				tbl[1] = tbl[1] + lp_reward
				current_lp = tbl[1]
				if consec_days % 5 == 0 then
					tbl[4] = tbl[4] + 2
				end
				current_spins = tbl[4]
			else --didn't log in yesterday, reset counter
				tbl[2] = 1
				consec_days = 1
				lp_reward = 600
				tbl[1] = tbl[1] + lp_reward
				current_lp = tbl[1]
				tbl[3] = day
				current_spins = tbl[4]
			end
		end

		local loyaltystats = Instance.new("Folder")
		loyaltystats.Name = "LoyaltyStats"

		lp.Value = current_lp
		local oldlp = Instance.new("IntValue",loyaltystats)
		oldlp.Name = "OldLP"
		oldlp.Value = current_lp
		
		local spins = Instance.new("IntValue",loyaltystats)
		spins.Name = "Spins"
		spins.Value = current_spins

		oldspins.Name = "OldSpins"

		local rewardstat = Instance.new("IntValue",loyaltystats)
