function RepeatGet(pd, key, successes, tries)
	local success, reason, returnvalue
	local i = 0
	repeat
		if i ~= 0 then
			wait(3)
		end
		success, reason = pcall(function()
			returnvalue = pd:GetAsync(key)
		end)
		i = i + 1
	until success or i >= (tries or 10)
	
	if successes then
		table.insert(successes,success)
		if debugmode or not success then print(reason) end
	end
	
	return returnvalue, success, reason
end

	local function GetLoyaltyData(tbl)
		local changes = false
		local timedata = game.ReplicatedStorage.IntroGUIData
		local day = timedata.CurrentRealLifeDay.Value
		local endtime = timedata.HolidayStartTime.Value + (day * 86400)
		local timeleft = endtime - timedata.CurrentTime.Value
		local timeleftstring = math.floor(timeleft/3600).."h "..math.floor((timeleft%3600)/60).."m"
		
		local current_lp = tbl[1]
		local lp_reward = 0
		local consec_days = tbl[2]
		local current_spins = tbl[4]
		local firstlogintoday = false
		local lastdayloggedin = tbl[3]
		
		if tbl[3] ~= day then --hasn't logged in today
			firstlogintoday = true
			changes = true
			if day - tbl[3] <= 2 then --logged in yesterday or 2 days before, offer rewards for consec. logins
				tbl[3] = day
				tbl[2] = tbl[2] + 1
				consec_days = tbl[2]
				if tbl[2] >= 5 then
					lp_reward = 1000
				else
					lp_reward = 500 + tbl[2] * 100
				end
				tbl[1] = tbl[1] + lp_reward
				current_lp = tbl[1]
				if consec_days % 5 == 0 then
					tbl[4] = tbl[4] + 2
				end
				current_spins = tbl[4]
			else --didn't log in yesterday, reset counter
				tbl[2] = 1
				consec_days = 1
				lp_reward = 600
				tbl[1] = tbl[1] + lp_reward
				current_lp = tbl[1]
				tbl[3] = day
				current_spins = tbl[4]
			end
		end

		local loyaltystats = Instance.new("Folder")
		loyaltystats.Name = "LoyaltyStats"

		lp.Name = "LP"
		lp.Value = current_lp
		local oldlp = Instance.new("IntValue",loyaltystats)
		oldlp.Name = "OldLP"
		oldlp.Value = current_lp
		
		local spins = Instance.new("IntValue",loyaltystats)
		spins.Name = "Spins"
		spins.Value = current_spins
		local oldspins = Instance.new("IntValue",loyaltystats)
		oldspins.Name = "OldSpins"
		oldspins.Value = current_spins
		
		local lastday = Instance.new("IntValue",loyaltystats)
		lastday.Name = "LastDayLoggedIn"
		lastday.Value = lastdayloggedin
		
		--store data that is only used to determine what's displayed on GUI
		local rewardstat = Instance.new("IntValue",loyaltystats)
		rewardstat.Name = "LPReward"
		rewardstat.Value = lp_reward

		playerloginday.Value = day

function GetPrize(plr,tier)
	if tier == 1 then
		
		local chance = math.random(1,11)
		if chance <= 7 then --furniture 200-520
			return GetFurniture(plr,200,520,tier)
		else --cash 250-400 ($25 incs.)
			return GetCash(plr,225 + 25 * math.random(1,9),tier)
		end
		
	elseif tier == 2 then
		
		local chance = math.random(1,21)
		local possiblepets = GetItems(plr,"pets",250,1200)
		local possiblegear = GetItems(plr,"gear",150,900)
		if chance >= 12 and chance <= 18 and #possiblegear > 0 then
			return GetPermanentItem(plr,possiblegear[math.random(1,#possiblegear)],tier)
		elseif chance >= 19 and chance <= 21 and #possiblepets > 0 then
			return GetPermanentItem(plr,possiblepets[math.random(1,#possiblepets)],tier)
		else
			return GetFurniture(plr,500,950,tier)
		end
		
	elseif tier == 3 then
		
		local chance = math.random(1,10)
		local possiblegear = GetItems(plr,"gear",1000,3000)
		if chance >= 5 and chance <= 9 and #possiblegear > 0 then
			return GetPermanentItem(plr,possiblegear[math.random(1,#possiblegear)],tier)
		elseif chance == 10 then
			return GetCash(plr,750 + 250 * math.random(1,7),tier)
		else
			return GetFurniture(plr,1000,4200,tier)
		end
		
	elseif tier == 4 then
		return GetCash(plr,100000,tier)
	end
end

			tier = 2
		elseif tierchance >= 9351 and tierchance <= 9999 then
			tier = 3
		elseif tierchance == 10000 then
			tier = 4
		end
		print("Tier Chance: "..tierchance.." ... Prize Tier: "..tier)
		
		local availableindicies = {}
		
		for i,itemtable in pairs(tbl) do
			if itemtable["PrizeTier"] == tier then
				table.insert(availableindicies,i)
			end
		end
		
		data.LoyaltyStats.Spins.Value = data.LoyaltyStats.Spins.Value - 1
		
		local index = availableindicies[math.random(1,#availableindicies)] --winner!
		--now a function that will wait for the client's wheel to stop spinning to award the prize
		WaitForWheelStop(plr,tbl[index])
		
		print("Index "..index.." selected, player will win "..tbl[index]["ItemName"])
		return index
	end
