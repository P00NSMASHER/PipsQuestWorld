game.ReplicatedStorage.RemoteFunctions.SendSprayPaint.OnClientEvent:connect(function(plr,sprayid, mousepos, surface, target)
	if target == nil then return end
	local canspray=false
	if (surface==Enum.NormalId.Front or surface==Enum.NormalId.Back) and target.Size.x>=spraysize and target.Size.y>=spraysize then
		canspray=true
	elseif (surface==Enum.NormalId.Left or surface==Enum.NormalId.Right) and target.Size.y>=spraysize and target.Size.z>=spraysize then
		canspray=true
	elseif (surface==Enum.NormalId.Top or surface==Enum.NormalId.Bottom) and target.Size.x>=spraysize and target.Size.z>=spraysize then
		canspray=true
	end
	if canspray then
		local p=Instance.new("Part")
		local d=Instance.new("Decal")
		d.Texture=baseid..tostring(sprayid-1)
		d.Face=surface
		d.Parent=p
		p.Name="!SPRAY_"..plr.Name
		p.formFactor="Custom"
		p.Anchored=false
		p.CanCollide=false
		p.Transparency=1
		p.CustomPhysicalProperties = PhysicalProperties.new(0.01,1,1)
		if surface==Enum.NormalId.Front or surface==Enum.NormalId.Back then
			p.Size=Vector3.new(spraysize,spraysize,.2)
		elseif surface==Enum.NormalId.Left or surface==Enum.NormalId.Right then
			p.Size=Vector3.new(.2,spraysize,spraysize)
		elseif surface==Enum.NormalId.Top or surface==Enum.NormalId.Bottom then
			p.Size=Vector3.new(spraysize,.2,spraysize)
		end
		local w=Instance.new("Weld")
		w.Part0=target
		w.Part1=p
		local cf=CFrame.new(target.CFrame:pointToObjectSpace(mousepos))
		w.C0=cf
		w.C1=CFrame.new(0,0,0)
		w.Parent=target
		p.CFrame=target.CFrame:toWorldSpace(cf)		--this is here to position the spray other wise it will be at 0,0,0 if the target is anchored
		game:GetService("Debris"):AddItem(p,60)
		p.Parent=game.Workspace
	end
end)

function commanumber(num)
	local num = tostring(num)
	local i = 0
	local finalstring = ""
	for n in string.gmatch(num,"%d") do
		finalstring = finalstring..n
		i = i + 1
		if (#num - i) % 3 == 0 and i ~= #num then
			finalstring = finalstring..","
		end
	end

					local marker = Instance.new("Folder",restrict)
					marker.Name = targetplr.Name
					return true
				end
			else
				return false,"Can't whitelist yourself"
			end
		else
			return false,"Player not in server"
		end
	end
end


function getFurnitureCostMarker(stats,servobj)
	local furnbought
	local id = servobj.ItemID.Value
	if stats.FurnitureCost:FindFirstChild(id) then
		furnbought = stats.FurnitureCost:FindFirstChild(id)
	else
		furnbought = Instance.new("IntValue")
		furnbought.Value = 0
		furnbought.Name = id
		furnbought.Parent = stats.FurnitureCost
	end
	local pending
	if furnbought:FindFirstChild("PendingSales") then
		pending = furnbought.PendingSales
	else
		pending = Instance.new("IntValue")
		pending.Value = 0
		pending.Name = "PendingSales"
		pending.Parent = furnbought
	end
	return furnbought,pending
end


function getOwnedObjectCount(plr,itemid)
	local count = 0
	local data = game.ReplicatedStorage.PlayerData:FindFirstChild(plr.userId)
	if data then
		for _,v in pairs(data.Objects:GetChildren()) do
			if tonumber(v.Name) == itemid then
				count = count + 1
			end
		end
	end
	return count
end

function RemoveFurnitureItem(plr,model) --referenced later down
	local house = workspace:FindFirstChild("!home_"..plr.Name)
	local stats = game.ReplicatedStorage.PlayerData:FindFirstChild(plr.userId)
	if house and stats and model.Parent == house then
		local itemserv = game.ReplicatedStorage.Objects:FindFirstChild(model.Name)
		if itemserv then
			local furnbought,pending = getFurnitureCostMarker(stats,itemserv)
			furnbought.Value = furnbought.Value + 1
			model:Destroy()
			if getOwnedObjectCount(plr,itemserv.ItemID.Value) + furnbought.Value <= 0 then

game.ReplicatedStorage.RemoteFunctions.GetFurniture.OnServerInvoke = function(plr,objname)
	local house = workspace:FindFirstChild("!home_"..plr.Name)
	local stats = game.ReplicatedStorage.PlayerData:FindFirstChild(plr.userId)
	if plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") and house and house.Floor.Editing.Value == true and stats then
		local torso = plr.Character.HumanoidRootPart
		local ray = Ray.new((torso.CFrame + torso.CFrame.lookVector * 10).p, (((torso.CFrame + torso.CFrame.lookVector * 10).p + Vector3.new(0,-15,0)) - (torso.CFrame + torso.CFrame.lookVector * 10).p).unit*15)
		local ignoredparts = {}
		for i,v in pairs(house:GetChildren()) do
			if v:IsA("Model") then
				table.insert(ignoredparts,v)
			end
		end
		--check if ray is inside house
		local findpart = workspace:FindPartOnRayWithIgnoreList(ray,ignoredparts)
		if findpart == nil or findpart.Parent.Name ~= house.Name or not findpart.Name:match("Floor") then
			return false,"Please stand facing some empty space inside your house, then try again."
		end
		local fcount = getFurnitureCount(plr)
		if fcount == 450 then
			game.ReplicatedStorage.RemoteFunctions.NotificationForClient:FireClient(plr, "Notice: You are approaching the item limit for your house. You have placed "..fcount.."/500 items.")
		end
		if fcount >= 500 then
			--hoarder!!
			return false,"Sorry, you cannot fit more than 1000 items in your home!"
		end
		
		
		local components = (torso.CFrame + torso.CFrame.lookVector * 10).p
		local itemserv = game.ReplicatedStorage.Objects:FindFirstChild(objname)
		if itemserv then
			local item = itemserv:clone()
			for _,v in pairs(item:GetChildren()) do
				if v.ClassName:match("Value") and v.Name ~= "PaintColor" and v.Name ~= "RotValue" and v.Name ~= "ItemID" then
					v:Destroy()
				end
			end
			item.Parent = house
			
			item:SetPrimaryPartCFrame(house.Floor.CFrame)
			local boxsize,boxcframe = require(game.ReplicatedStorage.Modules.GetBoundingBox)(item)
			local bottomboundary = boxcframe.Y - boxsize.Y / 2
			local yoffset = item.PrimaryPart.Position.Y - bottomboundary + house.Floor.Size.Y / 2
			local origpos = item.PrimaryPart.Position
			item:SetPrimaryPartCFrame(item.PrimaryPart.CFrame + (Vector3.new(math.floor(components.X),yoffset,math.floor(components.Z))) - Vector3.new(origpos.X,0,origpos.Z)) -- orig CFrame + (newpos - oldpos)

			local tempval = Instance.new("BoolValue")
			tempval.Name = "temp"
			tempval.Parent = item
			local furnbought,pending = getFurnitureCostMarker(stats,itemserv)
			furnbought.Value = furnbought.Value - 1
			if getOwnedObjectCount(plr,itemserv.ItemID.Value) + pending.Value + furnbought.Value < 0 then
				stats.FurnitureCost.Value = stats.FurnitureCost.Value + itemserv.ItemCost.Value
			end
			getFurnitureCount(plr)
			return true
		end
		
	end
end

function GetPrize(plr,tier)
	if tier == 1 then
		
		local chance = math.random(1,11)
		if chance <= 7 then --furniture 200-520
			return GetFurniture(plr,200,520,tier)
		else --cash 250-400 ($25 incs.)
			return GetCash(plr,225 + 25 * math.random(1,9),tier)
		end
		
	elseif tier == 2 then
		
		local chance = math.random(1,21)
		local possiblepets = GetItems(plr,"pets",250,1200)
		local possiblegear = GetItems(plr,"gear",150,900)
		if chance >= 12 and chance <= 18 and #possiblegear > 0 then
			return GetPermanentItem(plr,possiblegear[math.random(1,#possiblegear)],tier)
		elseif chance >= 19 and chance <= 21 and #possiblepets > 0 then
			return GetPermanentItem(plr,possiblepets[math.random(1,#possiblepets)],tier)
		else
			return GetFurniture(plr,500,950,tier)
		end
		
	elseif tier == 3 then
		
		local chance = math.random(1,10)
		local possiblegear = GetItems(plr,"gear",1000,3000)
		if chance >= 5 and chance <= 9 and #possiblegear > 0 then
			return GetPermanentItem(plr,possiblegear[math.random(1,#possiblegear)],tier)
		elseif chance == 10 then
			return GetCash(plr,750 + 250 * math.random(1,7),tier)
		else
			return GetFurniture(plr,1000,4200,tier)
		end
		
	elseif tier == 4 then
		return GetCash(plr,100000,tier)
	end
end
